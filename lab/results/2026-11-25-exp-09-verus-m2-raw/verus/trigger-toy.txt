# Experiment #9 run 10: why statement.rs needed a named per-instance body.
# Toy crates checked on the bench host (Verus 0.2026.11.02.8bf68ee,
# untimed, nice -n 10, CPUs 4-7, between 15:20:28Z and 15:24:10Z), outside
# the proof edit loop, so not in invocations.tsv.  The toy reproduces the
# shape of run 3's main_theorem_statement: an outer forall whose trigger
# (dummy(x), like sem(0, r, s, p)) occurs nowhere in its body outside a
# quantifier, and an exists in the body.

## toy 1: statement behind a spec fn, proved by an identical assert forall
pub open spec fn dummy(x: nat) -> bool { true }
pub open spec fn f(y: nat, x: nat) -> bool { y > x }
pub open spec fn st() -> bool { forall|x: nat| #![trigger dummy(x)] x > 3 ==> exists|y: nat| #[trigger] f(y, x) }
pub proof fn t() ensures st() {
    assert forall|x: nat| #![trigger dummy(x)] x > 3 ==> exists|y: nat| #[trigger] f(y, x) by { assert(f(x+1, x)); }
}
-> error: postcondition not satisfied; verification results:: 0 verified, 1 errors

## toy 3: the body named by a spec fn and used as the outer trigger (the run-10 statement.rs idiom)
pub open spec fn body(x: nat) -> bool { x > 3 ==> exists|y: nat| #[trigger] f(y, x) }
pub open spec fn st() -> bool { forall|x: nat| #[trigger] body(x) }
pub proof fn t() ensures st() {
    assert forall|x: nat| #[trigger] body(x) by { assert(f(x+1, x)); }
}
-> verification results:: 1 verified, 0 errors

# Reading: to refute the negated goal, Z3 must instantiate the asserted
# forall at the skolem constants of the goal, which needs a ground term
# matching the trigger.  dummy(x0) (sem(0, r0, s0, p0)) never appears;
# body(x0) (main_theorem_at(r0, s0, p0)) does.
